#include <stdio.h>

int main()
{
char name[41][50];
int booked[41] = {0};
int sno[41] = {0};
int count = 0, choice, seat, i;

do  
{  
    printf("\n--- BUS TICKET RESERVATION ---\n");  
    printf("1. Book a Seat\n");  
    printf("2. View Seat Map\n");  
    printf("3. Search Booking\n");  
    printf("4. Exit\n");  

    printf("Enter choice: ");  
    scanf("%d", &choice);  

    switch(choice)  
    {  
        case 1:  
            printf("Enter passenger name: ");  
            scanf(" %49[^\n]", name[1]);  

            printf("Enter seat number (1-40): ");  
            scanf("%d", &seat);  

            if(seat < 1 || seat > 40)  
                printf("Invalid seat number!\n");  

            else if(booked[seat] == 1)  
                printf("Seat already booked!\n");  

            else  
            {  
                printf("Enter passenger name: ");  
                scanf(" %49[^\n]", name[seat]);  

                booked[seat] = 1;  
                count++;  
                sno[seat] = count;  

                printf("Seat booked successfully!\n");  
                printf("Serial Number: %d\n", count);  
            }  
            break;  

        case 2:  
            printf("\n--- BUS SEAT MAP ---\n");  
            printf("B = Booked, F = Free\n");  

            for(i = 1; i <= 40; i++)  
            {  
                if(booked[i] == 1)  
                    printf("%02d[B]  ", i);  
                else  
                    printf("%02d[F]  ", i);  

                if(i % 4 == 0)  
                    printf("\n");  
            }  
            break;  

        case 3:  
            printf("Enter seat number to search: ");  
            scanf("%d", &seat);  

            if(seat < 1 || seat > 40)  
                printf("Invalid seat number!\n");  

            else if(booked[seat] == 1)  
            {  
                printf("Booking Found!\n");  
                printf("S.No: %d\n", sno[seat]);  
                printf("Passenger: %s\n", name[seat]);  
                printf("Seat Number: %d\n", seat);  
            }  
            else  
                printf("Booking not found!\n");  
            break;  

        case 4:  
            printf("Exiting program. Thank you!\n");  
            break;  

        default:  
            printf("Invalid choice!\n");  
    }  

} while(choice != 4);  

return 0;

}
